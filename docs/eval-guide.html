10 案例评估搭建手册

给「Amazon 智能竞品分析系统」补上一套能复现、能写进简历的离线评估。

手写代码约 2.5 小时 跑批 40–100 分钟（可后台） 10 个案例：7 正向 + 3 负向 产出 evals/ + report.md

这份手册怎么用

  1. 10 步，每步末尾都有一个「验收」。验收没过，不要往下走——尤其是第 4、5 步， 它们的作用就是防止你花两小时跑出 10 份废数据。
  2. 每步给的是完整文件：建同名文件，把代码粘进去，执行验收命令。
  3. 代码里没有需要你改的地方。唯一需要你动脑的是第 3 步：去 Amazon 页面人工记录预期值。
  4. 左侧可以勾选进度，关掉浏览器再打开还在。
  5. 文件就在项目里：docs/eval-guide.html，双击用浏览器打开即可，不依赖网络。

为什么值得做这件事

没有评估的 Agent 项目，面试时只能讲「我实现了什么」。有评估，才能讲 「它有多好、哪里不行、我怎么知道的」。这三句话是面试官区分 「跟着教程做过」和「自己负责过一个系统」的地方。

这份评估最终给你三样东西：

  • 可复现的指标：字段准确率、竞品有效性、任务成功率、耗时、成本，一条命令重跑。
  • 失败模式清单：哪类输入会挂、挂在哪一步、失败时系统是报错还是硬编一个结果。
  • 代价画像：一次分析要多久、花多少钱、钱花在哪个环节。

指标口径

先把口径定死，后面所有数字才有意义。这六个指标就是你要在简历/面试里引用的全部。

指标口径怎么算
商品字段准确率 目标商品的 5 个关键字段：ASIN、品牌、价格、评分、评论数 与你人工记录的预期值逐项比对。价格允许 ±5%，评分 ±0.1，评论数 ±max(50, 10%)。 只统计你填了预期值的字段；留空 = 不计分。
竞品有效性得分 前 3 个竞品，每个 6 项检查 ASIN 是 10 位 / 不是目标自己 / 品牌与目标不同 / 标题命中类目关键词 / 价格评分评论齐全 / 有来源链接。6 项通过率高 = 竞品找得准。
信息完整度 落库商品的 8 个字段 name、title、asin、brand、price、currency、rating、reviews 的填充率。
任务成功率 端到端是否达成预期 正向：completed + 报告非空且不是模拟 + 有效竞品数达标。 负向：failed + 有错误信息。
耗时 一次分析的真实墙钟时间 从 run_analysis 开始到返回（含竞品发现、并行分析、报告生成）。
API 成本 token + 外部调用 input/output token × 单价 + embedding token × 单价 + 搜索次数 × 单价 + 抓取任务数 × 单价。单价在 pricing.json。

def load_json(path: Path) -> Any:
    with path.open("r", encoding="utf-8") as file:
        return json.load(file)


def load_cases() -> list[dict]:
    cases = load_json(CASES_PATH)

    if not isinstance(cases, list) or not cases:
        raise ValueError("cases.json 必须是非空数组")

    ids = [case.get("id") for case in cases]

    if any(not case_id for case_id in ids):
        raise ValueError("每个案例都必须有 id")

    if len(ids) != len(set(ids)):
        raise ValueError("cases.json 存在重复 id")

    return cases


def load_pricing() -> dict:
    return load_json(PRICING_PATH)


def write_json(path: Path, data: Any) -> None:
    """先写临时文件再替换，避免中途退出留下半个 JSON。"""

    path.parent.mkdir(parents=True, exist_ok=True)
    temp_path = path.with_suffix(path.suffix + ".tmp")

    with temp_path.open("w", encoding="utf-8") as file:
        json.dump(data, file, ensure_ascii=False, indent=2)
        file.flush()
        os.fsync(file.fileno())

    temp_path.replace(path)


def setup_logging(case_id: str) -> Path:

    root.setLevel(logging.INFO)

    for handler in list(root.handlers):

    console = logging.StreamHandler()
    console.setFormatter(formatter)

    file_handler = logging.FileHandler(log_path, encoding="utf-8")
    file_handler.setFormatter(formatter)

    root.addHandler(console)

class UsageTracker:
    """统计 LLM token 与外部工具调用次数。"""

    def __init__(self) -> None:
        self._lock = threading.Lock()
        self._installed = False
        self.data = {
            "prompt_tokens": 0,
            "completion_tokens": 0,
            "embedding_tokens": 0,
            "llm_calls": 0,
            "embedding_calls": 0,
            "usage_missing": 0,
            "tavily_calls": 0,
            "brightdata_jobs": 0,
        }

    def add(self, key: str, value: int = 1) -> None:
        with self._lock:
            self.data[key] += value

    def snapshot(self) -> dict:
        with self._lock:
            return dict(self.data)

    def install(self) -> None:
        """给 OpenAI SDK 和外部工具打补丁，只影响当前进程。"""

        if self._installed:
            return

        from openai.resources.chat.completions import Completions
        from openai.resources.embeddings import Embeddings

        tracker = self

        original_chat = Completions.create

        def chat_create(self, *args, **kwargs):
            response = original_chat(self, *args, **kwargs)
            usage = getattr(response, "usage", None)

            with tracker._lock:
                tracker.data["llm_calls"] += 1
                if usage is None:
                    tracker.data["usage_missing"] += 1
                else:
                    tracker.data["prompt_tokens"] += (
                        getattr(usage, "prompt_tokens", 0) or 0
                    )
                    tracker.data["completion_tokens"] += (
                        getattr(usage, "completion_tokens", 0) or 0
                    )

            return response

        Completions.create = chat_create

        original_embed = Embeddings.create

        def embed_create(self, *args, **kwargs):
            response = original_embed(self, *args, **kwargs)
            usage = getattr(response, "usage", None)

            with tracker._lock:
                tracker.data["embedding_calls"] += 1

                if usage is None:
                    tracker.data["usage_missing"] += 1
                    total = 0
                else:
                    total = getattr(usage, "total_tokens", None)

                    if total is None:
                        total = (
                            getattr(usage, "prompt_tokens", 0) or 0
                        ) + (
                            getattr(usage, "completion_tokens", 0) or 0
                        )

                tracker.data["embedding_tokens"] += total or 0

            return response

        Embeddings.create = embed_create

        try:
            import app.tools.web_search as web_search

            original_search = web_search.tavily_client.search

            def counted_search(*args, **kwargs):
                tracker.add("tavily_calls")
                return original_search(*args, **kwargs)

            web_search.tavily_client.search = counted_search
        except Exception:
            logger.exception("Tavily 计数补丁安装失败")

        try:
            import app.analyzers.product_review_pipeline as review_pipeline

            original_reviews = review_pipeline.collect_amazon_reviews

            def counted_reviews(*args, **kwargs):
                tracker.add("brightdata_jobs")
                return original_reviews(*args, **kwargs)

            review_pipeline.collect_amazon_reviews = counted_reviews
        except Exception:
            logger.exception("BrightData 计数补丁安装失败")

        self._installed = True


def normalize(value: Any) -> str:
    return str(value or "").strip().casefold()


def as_float(value: Any) -> float | None:
    try:
        return float(value)
    except (TypeError, ValueError):
        return None


def as_int(value: Any) -> int | None:
    number = as_float(value)
    return int(number) if number is not None else None


def target_field_checks(
    expected: dict,
    target: dict | None,
) -> dict[str, bool | None]:
    """逐字段比对目标商品。返回 None 表示该字段没有预期值，不参与计分。"""

    checks: dict[str, bool | None] = {}

    expected_asin = normalize(expected.get("target_asin"))
    actual_asin = normalize((target or {}).get("asin"))

    if not expected_asin:
        checks["asin"] = None
    else:
        checks["asin"] = actual_asin == expected_asin

    expected_brand = normalize(expected.get("brand"))
    actual_brand = normalize((target or {}).get("brand"))

    if not expected_brand:
        checks["brand"] = None
    else:
        checks["brand"] = bool(actual_brand) and (
            expected_brand in actual_brand or actual_brand in expected_brand
        )

    expected_price = as_float(expected.get("price"))
    actual_price = as_float((target or {}).get("price"))

    if expected_price is None:
        checks["price"] = None
    else:
        checks["price"] = (
            actual_price is not None
            and abs(actual_price - expected_price)
            / max(abs(expected_price), 0.01)
            <= PRICE_TOLERANCE
        )

    expected_rating = as_float(expected.get("rating"))
    actual_rating = as_float((target or {}).get("rating"))

    if expected_rating is None:
        checks["rating"] = None
    else:
        checks["rating"] = (
            actual_rating is not None
            and abs(actual_rating - expected_rating) <= RATING_TOLERANCE
        )

    expected_reviews = as_int(expected.get("reviews"))
    actual_reviews = as_int((target or {}).get("reviews"))

    if expected_reviews is None:
        checks["reviews"] = None
    else:
        checks["reviews"] = (
            actual_reviews is not None
            and abs(actual_reviews - expected_reviews)
            <= max(REVIEWS_FLOOR, abs(expected_reviews) * REVIEWS_TOLERANCE)
        )

    return checks


def rate_checks(checks: dict[str, bool | None]) -> float | None:
    scored = [value for value in checks.values() if value is not None]

    if not scored:
        return None

    return sum(1 for value in scored if value) / len(scored)


def competitor_checks(
    competitor: dict,
    target: dict | None,
    expected: dict,
) -> dict[str, bool | None]:
    text = f'{competitor.get("title") or ""} {competitor.get("name") or ""}'.lower()
    keywords = [
        str(keyword).lower()
        for keyword in expected.get("category_keywords", [])
        if keyword
    ]

    target_asin = normalize((target or {}).get("asin"))
    target_brand = normalize((target or {}).get("brand"))
    competitor_brand = normalize(competitor.get("brand"))

    # 品牌写法常带后缀（Samsung / Samsung Electronics），用包含关系判断同一品牌
    same_brand = (
        bool(target_brand)
        and bool(competitor_brand)
        and (
            competitor_brand in target_brand
            or target_brand in competitor_brand
        )
    )

    checks: dict[str, bool | None] = {
        "asin_format": (
            bool(competitor.get("asin"))
            and len(str(competitor["asin"]).strip()) == 10
        ),
        "not_target": (
            bool(competitor.get("asin"))
            and normalize(competitor.get("asin")) != target_asin
        ),
        "different_brand": (
            None
            if not target_brand
            else bool(competitor_brand) and not same_brand
        ),
        "category": (
            any(keyword in text for keyword in keywords) if keywords else None
        ),
        "fields_present": all(
            competitor.get(field) is not None
            for field in ("price", "rating", "reviews")
        ),
        "source_present": bool(competitor.get("sources")),
    }

    return checks


def competitor_is_valid(checks: dict[str, bool | None]) -> bool:
    for field in ("asin_format", "not_target", "different_brand", "category"):
        if checks[field] is False:
            return False

    return bool(checks["asin_format"])


def score_competitors(
    competitors: list[dict],
    target: dict | None,
    expected: dict,
) -> dict:
    scored = competitors[:SCORED_COMPETITORS]

    if not scored:
        return {"rate": 0.0, "valid_count": 0, "details": []}

    details = []
    rates = []
    valid_count = 0

    for competitor in scored:
        checks = competitor_checks(competitor, target, expected)
        rate = rate_checks(checks)
        details.append({
            "asin": competitor.get("asin"),
            "name": competitor.get("name"),
            "checks": checks,
            "rate": rate,
        })

        if rate is not None:
            rates.append(rate)

        valid_count += int(competitor_is_valid(checks))

    return {
        "rate": sum(rates) / len(rates) if rates else 0.0,
        "valid_count": valid_count,
        "details": details,
    }


def completeness(products: list[dict]) -> float:
    if not products:
        return 0.0

    filled = sum(
        1
        for product in products
        for field in PRODUCT_FIELDS
        if product.get(field) not in (None, "")
    )

    return filled / (len(products) * len(PRODUCT_FIELDS))


def report_sections(markdown: str) -> list[str]:
    return [
        line.lstrip("#").strip()
        for line in markdown.splitlines()
        if line.startswith("#") and line.lstrip("#").strip()
    ]


def compute_cost(usage: dict, pricing: dict) -> dict:
    llm = (
        usage.get("prompt_tokens", 0)
        / 1_000_000
        * float(pricing.get("llm_input_per_1m", 0))
    ) + (
        usage.get("completion_tokens", 0)
        / 1_000_000
        * float(pricing.get("llm_output_per_1m", 0))
    )

    embedding = (
        usage.get("embedding_tokens", 0)
        / 1_000_000
        * float(pricing.get("embedding_per_1m", 0))
    )

    external = (
        usage.get("tavily_calls", 0)
        * float(pricing.get("tavily_per_search", 0))
    ) + (
        usage.get("brightdata_jobs", 0)
        * float(pricing.get("brightdata_per_job", 0))
    )

    return {
        "llm": round(llm, 6),
        "embedding": round(embedding, 6),
        "external": round(external, 6),
        "total": round(llm + embedding + external, 6),
    }

读这段代码时，只看这 4 个设计点

  1. PROJECT_ROOT = evals 的父目录：所有路径都从这里推导。 脚本在哪里启动都指向同一个项目根，不会出现「在 A 目录写进 A 的 reports」。
  2. 用 monkeypatch 统计用量，而不是改业务代码：评估不应该污染生产路径。 补丁只在当前进程有效，进程一退就没了——这也是「一个案例一个进程」的原因之一。
  3. threading.Lock 不是摆设：工作流用 ThreadPoolExecutor(max_workers=3) 并发分析竞品，计数不加锁会丢。
  4. 评分返回 None 表示「没有预期值，不计分」： 你还没填的字段不会被当成 0 分，避免自己坑自己。
为什么成本默认是 0： pricing.json 里的单价默认全 0。不确定真实单价时保持 0， 成本显示 $0.0000，远好过拍一个数字。面试被问就直说：token 数是真实测的，单价按公开价目表估。

怎么填预期值（全流程唯一需要你人工做的事）

  1. 浏览器打开 amazon.com，搜索 HUADISK 256GB NVMe SSD。
  2. 点开和查询最匹配的那个商品详情页。
  3. ASIN：看地址栏里 /dp/ 后面那 10 位，例如 /dp/B0XXXXXXXX → B0XXXXXXXX。
  4. 价格：页面上当前的美元价格，只写数字，例如 27.99。
  5. 评分：4.3 out of 5 stars → 4.3。
  6. 评论数：1,234 ratings → 1234。
  7. 在该案例的 note 末尾写上抓取日期，例如 … ｜ 2026-10-28 抓取。 价格和评论数会变，日期是你唯一的解释依据。
三条填数据的规矩： ① 不知道就留空——留空只是这个字段不计分，填错则直接判错； ② 三个负向案例（08/09/10）不需要 ASIN，它们预期的结果本来就是失败； ③ 别用「我印象里大概 30 块」这种数字，预期值必须是你在页面上亲眼看到的。

三个必须看懂的坑（代码里已经处理）

  1. MOCK_MODE 默认 true，而且在 import 时就取了值。 所以脚本在 import 之后强行改 False，并且立刻断言。不这样的话你会得到一份 「模拟报告 + completed 状态」，看起来一切正常，实际什么都没验证。

  3. 落库异常被吞掉：analysis_service 对 save_analysis_results 失败只记日志、不改状态。所以这里单独检查 「completed 但没有商品明细」，写进 notes。

"""批量运行全部案例：每个案例独立子进程，跑完自动汇总。

用法：
    python -m evals.run_all
    python -m evals.run_all --only case-01,case-02
    python -m evals.run_all --force
"""

from __future__ import annotations

import argparse
import subprocess
import sys
import time

from evals import framework


def parse_args() -> argparse.Namespace:
    parser = argparse.ArgumentParser(description="批量运行评估案例")
    parser.add_argument(
        "--only",
        default="",
        help="只跑指定案例，逗号分隔，例如 case-01,case-02",
    )
    parser.add_argument(
        "--force",
        action="store_true",
        help="已存在结果也重跑",
    )
    parser.add_argument(
        "--timeout",
        type=int,
        default=1800,
        help="单个案例超时秒数，默认 1800",
    )
    parser.add_argument(
        "--sleep",
        type=int,
        default=5,
        help="两个案例之间的间隔秒数，默认 5",
    )
    return parser.parse_args()


def selected_cases(only: str) -> list[dict]:
    cases = framework.load_cases()

    if not only.strip():
        return cases

    wanted = {item.strip() for item in only.split(",") if item.strip()}
    chosen = [case for case in cases if case["id"] in wanted]
    missing = wanted - {case["id"] for case in chosen}

    if missing:
        raise SystemExit(f"cases.json 里没有这些 id：{sorted(missing)}")

    return chosen


def main() -> int:
    args = parse_args()
    cases = selected_cases(args.only)

    framework.ensure_dirs()

    failed: list[str] = []
    skipped: list[str] = []

    for index, case in enumerate(cases, start=1):
        case_id = case["id"]
        result_path = framework.RESULTS_DIR / f"{case_id}.json"

        if result_path.exists() and not args.force:
            skipped.append(case_id)
            print(f"[{index}/{len(cases)}] 跳过 {case_id}：结果已存在")
            continue

        print(f"[{index}/{len(cases)}] 运行 {case_id}：{case['target_product']}")

        command = [
            sys.executable,
            "-m",
            "evals.run_eval",
            "--case",
            case_id,
            "--force",
        ]

        try:
            completed = subprocess.run(
                command,
                cwd=framework.PROJECT_ROOT,
                timeout=args.timeout,
            )
        except subprocess.TimeoutExpired:
            print(f"  -> 超时（>{args.timeout}s）")
            failed.append(case_id)
            continue

        if completed.returncode != 0:
            failed.append(case_id)

        if index < len(cases):
            time.sleep(args.sleep)

    print("\n完成。")
    print(f"跑完：{len(cases) - len(skipped)} 个，跳过：{len(skipped)} 个")

    if failed:
        print(f"未通过或异常：{', '.join(failed)}")

    print("下一步：python -m evals.summarize")
    return 0 if not failed else 1


if __name__ == "__main__":
    sys.exit(main())

为什么必须一个案例一个子进程

系统里有三处进程级缓存：网页搜索缓存、评论 single-flight 缓存、RAG 检索缓存。 同一个进程里跑第二个案例，很可能直接复用上一个案例的搜索结果和评论—— 看起来「很快、很便宜」，实际测的是一份被污染的数据。

先小后大

$python -m evals.run_all --only case-01,case-02

两个都正常，再全量：

$python -m evals.run_all

预计 40–100 分钟（单案例 3–10 分钟 × 10）。建议放后台跑，期间去写第 10 步的结论框架。

中断了不用重来： 已生成的 case-XX.json 会被自动跳过，重新执行同一条命令即可。 这就是 run_all.py 里 if result_path.exists() and not args.force 的作用。 想强制重跑某个案例，单独用 run_eval --case case-05 --force。

    path = framework.RESULTS_DIR / "summary.csv"

    with path.open("w", encoding="utf-8-sig", newline="") as file:
        writer = csv.DictWriter(file, fieldnames=CSV_FIELDS)
        writer.writeheader()

        for result in results:
            writer.writerow(row_of(result))

    print(f"已写入 {path}")


def aggregate(results: list[dict]) -> dict:
    positive = [
        result for result in results if result.get("case_type") == "positive"
    ]

    def average(values: list[float]) -> float | None:
        return mean(values) if values else None

    target_rates = [
        result["scores"]["target_field_accuracy"]
        for result in positive
        if result.get("scores", {}).get("target_field_accuracy") is not None
    ]

    durations = [result.get("duration_seconds", 0) for result in results]
    costs = [result.get("cost_usd", {}).get("total", 0) for result in results]

    return {
        "total": len(results),
        "positive": len(positive),
        "negative": len(results) - len(positive),
        "passed": sum(1 for result in results if result.get("success")),
        "success_rate": (
            sum(1 for result in results if result.get("success")) / len(results)
            if results
            else 0.0
        ),
        "target_accuracy": average(target_rates),
        "competitor_rate": average([
            result["scores"]["competitor"]["rate"]
            for result in positive
            if result.get("scores", {}).get("competitor")
        ]),
        "completeness": average([
            result["scores"]["completeness"]
            for result in results
            if result.get("scores", {}).get("completeness") is not None
        ]),
        "avg_duration": average(durations),
        "total_cost": sum(costs),
        "avg_cost": average(costs),
    }

面试怎么讲

这些是你做完评估后会被追问的问题。回答里的数字，只能来自你自己的 report.md。

问：你怎么评估这个系统的？

10 个案例，7 正向 3 负向，全部在真实模式下跑。正向看目标商品 5 个字段的准确率、 竞品 6 项有效性、任务成功率和信息完整度；负向看系统会不会在信息不足时硬编结果。 另外每个案例都记了耗时、token 和外部调用次数，换算成本。每个案例独立进程， 避免缓存污染。

问：10 个案例够吗？

不够做统计，但够定位失败模式。它是抽样体检，不是 benchmark——我在报告里 明确写了这一点，也写了价格会漂移、竞品有效性是启发式判定这些局限。

问：准确率怎么算的？

只统计我人工填了预期值的字段。价格允许 ±5%，评分 ±0.1，评论数 ±max(50, 10%)， 因为这三项本来就会随时间小幅波动。留空的字段不计分，不会被当成错。

问：成本怎么测的？

在评估进程里给 OpenAI SDK 的 create 和 Tavily 的检索打了补丁， 统计 token 与调用次数，再乘 pricing.json 里的单价。token 是真实的， 单价是公开价目表估的，所以我说的是「本地估算成本」，不是账单金额。

问：负向案例为什么算失败反而是好事？

负向案例测的是边界行为：无品牌、中文查询、不存在的商品。理想结果是系统明确报错， 而不是编一个 ASIN 出一份看起来正常的报告。后者在真实业务里更危险，因为用户看不出来是假的。

问：这套评估最大的局限是什么？

样本小、预期值靠人工录入会漂移、竞品有效性是启发式判定而不是人工打分、 成本是估算。这四条我都写进了报告的「不能证明什么」，没有包装成 benchmark。
